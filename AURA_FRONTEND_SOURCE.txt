/* =========================================
   AURA VOICE ASSISTANT

/* =========================================
   DOM ELEMENTS
========================================= */

const voiceButton =
    document.getElementById("voiceButton");

const startAssistantButton =
    document.getElementById("startAssistantButton");

const voiceVisualizer =
    document.getElementById("voiceVisualizer");

const assistantStatus =
    document.getElementById("assistantStatus");

const voiceInstruction =
    document.getElementById("voiceInstruction");

const transcript =
    document.getElementById("transcript");

const assistantResponse =
    document.getElementById("assistantResponse");

const clearTranscript =
    document.getElementById("clearTranscript");

const speakResponse =
    document.getElementById("speakResponse");

const stopSpeech =
    document.getElementById("stopSpeech");

const learnMoreButton =
    document.getElementById("learnMoreButton");

const commandCards =
    document.querySelectorAll(".command-card");


/* =========================================
   SPEECH RECOGNITION
========================================= */

const SpeechRecognition =
    window.SpeechRecognition ||
    window.webkitSpeechRecognition;


let recognition = null;

let isListening = false;


if (SpeechRecognition) {

    recognition = new SpeechRecognition();

    recognition.continuous = false;

    recognition.interimResults = true;

    recognition.lang = "en-IN";


    /* =====================================
       SPEECH START
    ===================================== */

    recognition.onstart = function () {

        isListening = true;

        voiceButton.classList.add("listening");

        voiceVisualizer.classList.add("listening");

        assistantStatus.textContent =
            "Listening...";

        voiceInstruction.textContent =
            "Speak your command now";

    };


    /* =====================================
       SPEECH RESULT
    ===================================== */

    recognition.onresult = function (event) {

        let finalTranscript = "";

        let interimTranscript = "";


        for (
            let i = event.resultIndex;
            i < event.results.length;
            i++
        ) {

            const text =
                event.results[i][0].transcript;


            if (event.results[i].isFinal) {

                finalTranscript += text;

            } else {

                interimTranscript += text;

            }

        }


        transcript.innerHTML = `
            <span>
                ${escapeHTML(
                    finalTranscript || interimTranscript
                )}
            </span>
        `;


        if (finalTranscript) {

            processCommand(
                finalTranscript.trim()
            );

        }

    };


    /* =====================================
       SPEECH END
    ===================================== */

    recognition.onend = function () {

        stopListening();

    };


    /* =====================================
       SPEECH ERROR
    ===================================== */

    recognition.onerror = function (event) {

        stopListening();


        let message =
            "Something went wrong. Please try again.";


        if (event.error === "not-allowed") {

            message =
                "Microphone permission was denied.";

        }


        if (event.error === "no-speech") {

            message =
                "I didn't hear anything. Please try again.";

        }


        showResponse(message);

    };

}


/* =========================================
   START LISTENING
========================================= */

function startListening() {

    if (!recognition) {

        showResponse(
            "Speech recognition is not supported in this browser. Please use Google Chrome or Microsoft Edge."
        );

        return;

    }


    try {

        recognition.start();

    } catch (error) {

        console.log(
            "Recognition is already running."
        );

    }

}


/* =========================================
   STOP LISTENING
========================================= */

function stopListening() {

    isListening = false;

    voiceButton.classList.remove("listening");

    voiceVisualizer.classList.remove("listening");

    assistantStatus.textContent =
        "Ready to listen";

    voiceInstruction.textContent =
        "Click the microphone and start speaking";

}


/* =========================================
   VOICE BUTTON
========================================= */

voiceButton.addEventListener(
    "click",
    function () {

        if (isListening) {

            recognition.stop();

        } else {

            startListening();

        }

    }
);


/* =========================================
   HERO START BUTTON
========================================= */

startAssistantButton.addEventListener(
    "click",
    function () {

        document
            .querySelector(".assistant-card")
            .scrollIntoView({
                behavior: "smooth",
                block: "center"
            });


        setTimeout(
            startListening,
            500
        );

    }
);


/* =========================================
   COMMAND PROCESSING
========================================= */

function processCommand(command) {

    const normalizedCommand =
        command.toLowerCase().trim();


    assistantStatus.textContent =
        "Processing command...";


    /* ==============================
       NAVIGATION
    ============================== */

    if (
        normalizedCommand.includes("go to home") ||
        normalizedCommand === "home"
    ) {

        showResponse(
            "Taking you to the home section."
        );

        document
            .getElementById("home")
            .scrollIntoView({
                behavior: "smooth"
            });

        return;

    }


    /* ==============================
       COMMANDS
    ============================== */

    if (
        normalizedCommand.includes("show commands") ||
        normalizedCommand.includes("open commands")
    ) {

        showResponse(
            "Opening the voice commands section."
        );

        document
            .getElementById("commands")
            .scrollIntoView({
                behavior: "smooth"
            });

        return;

    }


    /* ==============================
       HOW IT WORKS
    ============================== */

    if (
        normalizedCommand.includes("how does this work") ||
        normalizedCommand.includes("how it works")
    ) {

        showResponse(
            "AURA converts your speech into text, identifies your intent, performs an action, and provides voice feedback."
        );

        document
            .getElementById("how-it-works")
            .scrollIntoView({
                behavior: "smooth"
            });

        speakText(assistantResponse.textContent);

        return;

    }


    /* ==============================
       READ PAGE
    ============================== */

    if (
        normalizedCommand.includes("read this page") ||
        normalizedCommand.includes("read page")
    ) {

        const pageText =
            document.body.innerText;

        showResponse(
            "I'll read the page content for you."
        );

        speakText(
            pageText.substring(0, 2500)
        );

        return;

    }


    /* ==============================
       STOP SPEAKING
    ============================== */

    if (
        normalizedCommand.includes("stop speaking") ||
        normalizedCommand.includes("stop reading")
    ) {

        window.speechSynthesis.cancel();

        showResponse(
            "Speech stopped."
        );

        return;

    }


    /* ==============================
       SEARCH
    ============================== */

    if (
        normalizedCommand.startsWith("search for")
    ) {

        const searchQuery =
            normalizedCommand
                .replace("search for", "")
                .trim();


        if (searchQuery) {

            showResponse(
                `Searching for ${searchQuery}.`
            );

            speakText(
                `Searching for ${searchQuery}`
            );

            /*
             * Temporary frontend demonstration.
             *
             * In the backend phase this will
             * become a proper search API.
             */

            setTimeout(

function handleVoiceCommand(command) {
    const result = processCommand(command);

    // Update the response area.
    // Change this selector if your HTML uses a different ID.
    const responseElement = document.querySelector("#assistantResponse");

    if (typeof result === "string") {
        if (responseElement) {
            responseElement.textContent = result;
        }

        speakResponse(result);
        return;
    }

    if (!result || !result.action) {
        return;
    }

    switch (result.action) {
        case "readPage": {
            const pageText =
                document.querySelector("main")?.innerText ||
                document.body.innerText;

            speakResponse(pageText);
            break;
        }

        case "stopSpeaking":
            window.speechSynthesis.cancel();
            break;

        case "login":
            window.location.href = "/pages/login.html";
            break;

        case "register":
            window.location.href = "/pages/register.html";
            break;

        case "accessibility":
            window.location.href = "/pages/accessibility.html";
            break;

        case "settings":
            window.location.href = "/pages/settings.html";
            break;

        case "home":
            window.location.href = "/";
            break;

        default:
            speakResponse("Sorry, that action is not available yet.");
    }
}

    }

            spokenText

        totalCommands++

            'Error'

        resetAssistant()

        resetAssistant()

        recognition.start()

        console.log(error)

    }

export function processCommand(command) {
    const text = String(command || "")
        .toLowerCase()
        .trim()
        .replace(/[.!?,]+$/g, "")
        .replace(/\s+/g, " ");

    if (!text) {
        return "I didn't hear a command. Please try again.";
    }

    // Greetings
    if (/^(hi|hello|hey)( aura)?$/.test(text)) {
        return "Hello! I am AURA. How can I help you today?";
    }

    // General conversation
    if (text.includes("how are you") || text.includes("what are you doing")) {
        return "I'm doing well! I'm ready to help you navigate AURA.";
    }

    if (
        text.includes("who are you") ||
        text.includes("what is aura") ||
        text.includes("introduce yourself")
    ) {
        return "I am AURA, your accessibility assistant. I help you navigate pages and use supported voice commands.";
    }

    if (text.includes("thank you") || text.includes("thanks")) {
        return "You're welcome! I'm happy to help.";
    }

    // Stop speech
    if (
        text === "stop" ||
        text.includes("stop speaking") ||
        text.includes("stop reading") ||
        text.includes("be quiet")
    ) {
        return { action: "stopSpeaking" };
    }

    // Stop voice recognition
    if (text === "stop listening" || text === "stop listening to me") {
        return { action: "stopListening" };
    }

    // Read page
    if (
        text.includes("read this page") ||
        text.includes("read the page") ||
        text.includes("read aloud") ||
        text.includes("read out loud") ||
        text.includes("read dashboard") ||
        text.includes("read everything") ||
        text.includes("start reading")
    ) {
        return { action: "readPage" };
    }

    // Login
    if (
        text.includes("open login") ||
        text.includes("go to login") ||
        text.includes("login page") ||
        text.includes("sign in page")
    ) {
        return { action: "login" };
    }

    // Registration
    if (
        text.includes("open register") ||
        text.includes("open registration") ||
        text.includes("go to registration") ||
        text === "register" ||
        text.includes("sign up")
    ) {
        return { action: "register" };
    }

    // Accessibility
    if (
        text.includes("open accessibility") ||
        text.includes("accessibility options") ||
        text.includes("accessibility page") ||
        text.includes("go to accessibility")
    ) {
        return { action: "accessibility" };
    }

    // Settings
    if (
        text.includes("settings") ||
        text.includes("setting") ||
        text.includes("preferences")
    ) {
        return { action: "settings" };
    }

    // Dashboard
    if (
        text.includes("open dashboard") ||
        text.includes("go to dashboard") ||
        text.includes("dashboard page")
    ) {
        return { action: "dashboard" };
    }

    // Home
    if (
        text === "home" ||
        text.includes("go home") ||
        text.includes("home page") ||
        text.includes("open home")
    ) {
        return { action: "home" };
    }

    // Help
    if (
        text === "help" ||
        text.includes("what can you do") ||
        text.includes("available commands")
    ) {
        return 'You can say "read this page", "stop reading", "open login", "register", "open dashboard", "open settings", "open accessibility", or "go home".';
    }

    return `I heard "${command}". I don't recognize that command yet. Say "Help" to hear supported commands.`;
}

export function speak(message) {
    if (!("speechSynthesis" in window)) {
        console.error("Text-to-speech is not supported in this browser.");
        return false;
    }

    const text = String(message ?? "").trim();

    if (!text) {
        return false;
    }

    // Stop any previous speech before starting a new message.
    window.speechSynthesis.cancel();

    const utterance = new SpeechSynthesisUtterance(text);

    utterance.lang = "en-IN";
    utterance.rate = 0.95;
    utterance.pitch = 1;
    utterance.volume = 1;

    utterance.onerror = (event) => {
        console.error("AURA speech error:", event.error);
    };

    window.speechSynthesis.speak(utterance);

    return true;
}

export function stopSpeaking() {
    if ("speechSynthesis" in window) {
        window.speechSynthesis.cancel();
    }
}

import { speak, stopSpeaking } from "./speech.js";

const SpeechRecognition =

let recognition = null;
let isListening = false;

function showMessage(selector, message) {
    const element = document.querySelector(selector);

    if (element) {
        element.textContent = message;
    }
}

function handleVoiceAction(action) {
    switch (action) {
        case "readPage": {
            const content =
                document.querySelector("main") ||
                document.querySelector("#dashboard-app") ||
                document.querySelector("#assistant-app") ||
                document.querySelector("#settings-app") ||
                document.querySelector("#accessibility-app") ||
                document.body;

            const pageText = content?.innerText?.trim();

            if (pageText) {
                showMessage("#assistantResponse", "Reading the page.");
                speak(pageText);
            } else {
                speak("I couldn't find any page content to read.");
            }
            break;
        }

        case "stopSpeaking":
            stopSpeaking();
            showMessage("#assistantResponse", "Speech stopped.");
            break;

        case "stopListening":
            stopListening();
            showMessage("#assistantResponse", "Voice recognition stopped.");
            speak("Voice recognition stopped.");
            break;

        case "login":
            window.location.href = "/pages/login.html";
            break;

        case "register":
            window.location.href = "/pages/register.html";
            break;

        case "dashboard":
            window.location.href = "/pages/dashboard.html";
            break;

        case "accessibility":
            window.location.href = "/pages/accessibility.html";
            break;

        case "settings":
            window.location.href = "/pages/settings.html";
            break;

        case "home":
            window.location.href = "/";
            break;

        default:
            console.warn("Unsupported AURA action:", action);
            showMessage(
                "#assistantResponse",
                'That action is not supported yet. Say "help" for available commands.'
            );
            speak("That action is not supported yet.");
    }
}

function handleCommand(command) {
    showMessage("#userTranscript", command);

    let response;

    try {
        response = processCommand(command);
    } catch (error) {
        console.error("AURA command processing error:", error);
        showMessage(
            "#assistantResponse",
            "I couldn't process that command. Please try again."
        );
        return;
    }

    if (
        response &&
        typeof response === "object" &&
        response.action
    ) {
        handleVoiceAction(response.action);
        return;
    }

    const message = String(response ?? "");

    showMessage("#assistantResponse", message);

    if (message) {
        speak(message);
    }
}

export function startListening() {
    if (!SpeechRecognition) {
        showMessage(
            "#assistantResponse",
            "Speech recognition is not supported in this browser. Please use a recent version of Chrome."
        );
        return;
    }

    if (isListening) return;

    recognition = new SpeechRecognition();
    recognition.lang = "en-IN";
    recognition.continuous = false;
    recognition.interimResults = false;
    recognition.maxAlternatives = 1;

    recognition.onstart = () => {
        isListening = true;
        showMessage("#assistantResponse", "I'm listening. Please speak.");
    };

    recognition.onresult = (event) => {
        const result = event.results[event.resultIndex];

        if (!result || !result[0]) return;

        const command = result[0].transcript.trim();

        console.log("AURA heard:", command);
        handleCommand(command);
    };

    recognition.onerror = (event) => {
        console.error("AURA speech recognition error:", event.error);

        const messages = {
            "not-allowed":
                "Microphone access was denied. Allow microphone access in your browser settings.",
            "no-speech":
                "I didn't hear anything. Please try speaking again.",
            "audio-capture":
                "No microphone was detected. Check your microphone connection.",
            network:
                "Speech recognition encountered a network error. Please try again."
        };

        showMessage(
            "#assistantResponse",
            messages[event.error] ||
                `Speech recognition error: ${event.error}`
        );
    };

    recognition.onend = () => {
        isListening = false;
    };

    try {
        recognition.start();
    } catch (error) {
        isListening = false;
        console.error("Could not start AURA recognition:", error);
        showMessage(
            "#assistantResponse",
            "I couldn't start the microphone. Please try again."
        );
    }
}

export function stopListening() {
    if (recognition) {
        try {
            recognition.stop();
        } catch (error) {
            console.warn("Could not stop speech recognition:", error);
        }
    }

    isListening = false;
}

export function isVoiceListening() {
    return isListening;
}

function getPreferences() {
  try {
    return JSON.parse(localStorage.getItem("auraSettings") || "{}");
  } catch {
    return {};
  }
}

function getAccessibilityPreferences() {
  try {
    return JSON.parse(localStorage.getItem("auraAccessibility") || "{}");
  } catch {
    return {};
  }
}

function enabled(value, fallback = true) {
  if (value === undefined || value === null) return fallback;
  return value === true || value === "true";
}

function canUseVoice() {
  return enabled(getPreferences().voiceEnabled, true);
}

function canSpeak() {
  return enabled(getPreferences().speechEnabled, true) &&
    enabled(getAccessibilityPreferences().speechEnabled, true);
}

function cancelNavigation() {
  if (pendingNavigation !== null) {
    clearTimeout(pendingNavigation);
    pendingNavigation = null;
  }
}

function speakText(message) {
  if (!canSpeak()) {
    statusElement.textContent = "Spoken responses are disabled in your preferences.";
    return;
  }

  if (!("speechSynthesis" in window) ||
      typeof SpeechSynthesisUtterance === "undefined") {
    statusElement.textContent = "Spoken responses are not supported in this browser.";
    return;
  }

  window.speechSynthesis.cancel();
  const utterance = new SpeechSynthesisUtterance(String(message));
  utterance.lang = "en-IN";
  utterance.rate = 0.95;

  utterance.onstart = () => {
    statusElement.textContent = "AURA is speaking.";
  };

  utterance.onend = () => {
    statusElement.textContent = "Response finished. Ready for your next command.";
  };

  utterance.onerror = () => {
    statusElement.textContent = "Speech playback could not be completed.";
  };

  window.speechSynthesis.speak(utterance);
}

function respond(message, shouldSpeak = true) {
  lastResponse = message;
  responseElement.textContent = message;
  if (shouldSpeak) speakText(message);
}

function navigateAfterResponse(url, message) {
  cancelNavigation();
  respond(message);

  if (!canSpeak() ||
      !("speechSynthesis" in window) ||
      typeof SpeechSynthesisUtterance === "undefined") {
    window.location.href = url;
    return;
  }

  pendingNavigation = setTimeout(() => {
    window.location.href = url;
  }, Math.max(1800, Math.min(4000, message.length * 55)));
}

function stopSpeech() {
  cancelNavigation();
  if ("speechSynthesis" in window) window.speechSynthesis.cancel();
  statusElement.textContent = "Speech stopped.";
}

function handleAction(action) {
  const destinations = {
    login: ["/pages/login.html", "Opening the login page."],
    register: ["/pages/register.html", "Opening the registration page."],
    dashboard: ["/pages/dashboard.html", "Opening your dashboard."],
    accessibility: ["/pages/accessibility.html", "Opening accessibility tools."],
    settings: ["/pages/settings.html", "Opening settings."],
    home: ["/", "Opening the home page."]
  };

  if (action === "stopSpeaking") {
    stopSpeech();
    return;
  }

  if (action === "stopListening") {
    recognition?.stop();
    respond("Voice recognition stopped.");
    return;
  }

  if (action === "readPage") {
    const pageText = document.querySelector("main")?.innerText || document.body.innerText;
    respond(pageText);
    return;
  }

  const destination = destinations[action];

  if (destination) {
    navigateAfterResponse(destination[0], destination[1]);
  } else {
    respond("That action is not supported yet. Say help for available commands.");
  }
}

function processHomepageCommand(command) {
  recognizedElement.textContent = command;
  const response = processCommand(command);

  if (response && typeof response === "object" && response.action) {
    handleAction(response.action);
    return;
  }

  respond(String(response ?? ""));
}

function resetListeningUI() {
  isListening = false;
  startButton.disabled = false;
  orbButton.disabled = false;

  if (statusElement.textContent.startsWith("Listening")) {
    statusElement.textContent = "Finished listening. Try another command.";
  }
}

function startListening() {
  cancelNavigation();

  if (!canUseVoice()) {
    statusElement.textContent = "Voice commands are disabled in your preferences.";
    return;
  }

  const SpeechRecognition =
    window.SpeechRecognition || window.webkitSpeechRecognition;

  if (!SpeechRecognition) {
    statusElement.textContent = "Voice recognition is unavailable. Try a recent version of Chrome.";
    return;
  }

  if (isListening) {
    recognition?.stop();
    return;
  }

  recognition = new SpeechRecognition();
  recognition.lang = "en-IN";
  recognition.continuous = false;
  recognition.interimResults = false;
  recognition.maxAlternatives = 1;

  startButton.disabled = true;
  orbButton.disabled = true;
  statusElement.textContent = "Listening... Please speak now.";

  recognition.onstart = () => {
    isListening = true;
    statusElement.textContent = "Listening... Please speak now.";
  };

  recognition.onresult = (event) => {
    const command = event.results?.[event.resultIndex]?.[0]?.transcript?.trim();

    if (command) {
      console.log("AURA homepage heard:", command);
      processHomepageCommand(command);
    } else {
      statusElement.textContent = "I didn't catch that. Please try again.";
    }
  };

  recognition.onerror = (event) => {
    const messages = {
      "not-allowed": "Microphone access denied. Allow microphone access in your browser.",
      "service-not-allowed": "Microphone access is blocked by browser settings.",
      "no-speech": "No speech detected. Please try again.",
      "audio-capture": "No microphone was detected. Check your microphone connection.",
      network: "Voice recognition needs a working network connection."
    };

    statusElement.textContent =
      messages[event.error] || `Voice recognition error: ${event.error}. Please try again.`;
  };

  recognition.onend = resetListeningUI;

  try {
    recognition.start();
  } catch (error) {
    console.error("AURA speech recognition error:", error);
    resetListeningUI();
    statusElement.textContent = "Could not start voice recognition. Please try again.";
  }
}

/* =========================================
   START VOICE RECOGNITION
========================================= */

function startVoiceRecognition() {
    if (!recognition) {
        assistantStatus.textContent =
            'Voice recognition is not supported.';

        respond(
            'Please use a browser that supports speech recognition, such as Google Chrome.'
        );

        return;
    }

    if (isListening) {
        return;
    }

    if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
    }

    assistantStatus.textContent =
        'Listening... Speak now.';

    setListeningUI(true);

    try {
        recognition.start();
    } catch (error) {
        console.error(
            'Could not start speech recognition:',
            error
        );

        setListeningUI(false);

        assistantStatus.textContent =
            'Unable to start listening. Please try again.';
    }
}


/* =========================================
   SPEECH RECOGNITION EVENTS
========================================= */

if (recognition) {
    recognition.onstart = () => {
        assistantStatus.textContent =
            'Listening... Speak now.';

        setListeningUI(true);
    };

    recognition.onresult = (event) => {
        const command =
            event.results[0][0].transcript.trim();

        recognizedCommand.textContent = command;

        assistantStatus.textContent =
            'Command received.';

        setListeningUI(false);

        processCommand(command);
    };

    recognition.onerror = (event) => {
        console.error(
            'Speech recognition error:',
            event.error
        );

        setListeningUI(false);

        if (event.error === 'not-allowed') {
            assistantStatus.textContent =
                'Microphone access was denied.';

            respond(
                'Please allow microphone access in your browser settings and try again.'
            );

        } else if (event.error === 'no-speech') {
            assistantStatus.textContent =
                'No speech detected.';

            assistantResponse.textContent =
                'I could not hear you. Please try again.';

        } else {
            assistantStatus.textContent =
                'Unable to hear you.';

            assistantResponse.textContent =
                'Please try again.';
        }
    };

    recognition.onend = () => {
        setListeningUI(false);

        if (
            assistantStatus.textContent.startsWith('Listening')
        ) {
            assistantStatus.textContent =
                'Ready for your command';
        }
    };
}


/* =========================================
   COMMAND PROCESSOR
========================================= */

function processCommand(command) {
    const text = command.toLowerCase().trim();

    /* STOP SPEAKING */

    if (
        text.includes('stop speaking') ||
        text.includes('stop talking') ||
        text === 'stop'
    ) {
        stopSpeech();
        return;
    }

    /* READ PAGE */

    if (
        text.includes('read page') ||
        text.includes('read this page') ||
        text.includes('read the page')
    ) {
        readPage();
        return;
    }

    /* DASHBOARD */

    if (
        text.includes('open dashboard') ||
        text.includes('go to dashboard') ||
        text === 'dashboard'
    ) {
        navigateTo(
            'Opening your dashboard.',
            '/pages/dashboard.html'
        );
        return;
    }

    /* ACCESSIBILITY */

    if (
        text.includes('open accessibility') ||
        text.includes('accessibility')
    ) {
        navigateTo(
            'Opening accessibility controls.',
            '/pages/accessibility.html'
        );
        return;
    }

    /* SETTINGS */

    if (
        text.includes('open settings') ||
        text.includes('settings')
    ) {
        navigateTo(
            'Opening AURA settings.',
            '/pages/settings.html'
        );
        return;
    }

    /* LOGIN */

    if (
        text.includes('open login') ||
        text.includes('login')
    ) {
        navigateTo(
            'Opening the login page.',
            '/pages/login.html'
        );
        return;
    }

    /* REGISTER */

    if (
        text.includes('open register') ||
        text.includes('register') ||
        text.includes('sign up')
    ) {
        navigateTo(
            'Opening the registration page.',
            '/pages/register.html'
        );
        return;
    }

    /* HOME */

    if (
        text === 'home' ||
        text.includes('open home') ||
        text.includes('go home')
    ) {
        navigateTo(
            'Opening the home page.',
            '/'
        );
        return;
    }

    /* GO BACK */

    if (
        text === 'back' ||
        text.includes('go back') ||
        text.includes('previous page')
    ) {
        respond('Going back.');

        window.setTimeout(() => {
            window.history.back();
        }, 600);

        return;
    }

    /* GREETINGS */

    if (
        text.includes('hello') ||
        text.includes('hi aura') ||
        text === 'hi'
    ) {
        respond(
            'Hello! I am AURA. How can I help you?'
        );
        return;
    }

    /* HELP */

    if (
        text.includes('help') ||
        text.includes('what can you do')
    ) {
        respond(
            'I can open your dashboard, accessibility controls, settings, login and registration pages. I can also read this page aloud and stop my speech.'
        );
        return;
    }

    /* THANK YOU */

    if (
        text.includes('thank you') ||
        text.includes('thanks')
    ) {
        respond(
            "You're welcome! I'm here whenever you need help."
        );
        return;
    }

    /* UNKNOWN COMMAND */

    respond(
        `I heard "${command}". I do not have an action for that command yet. Try one of the available commands.`
    );
}

function respond(message, shouldSpeak = true) {
    lastResponse = message;

    assistantResponse.textContent = message;

    if (shouldSpeak) {
        speak(message);
    }
}


/* =========================================
   TEXT-TO-SPEECH
========================================= */

function speak(message) {
    if (!('speechSynthesis' in window)) {
        assistantStatus.textContent =
            'Text-to-speech is not supported in this browser.';

        return;
    }

    window.speechSynthesis.cancel();

    const speech =
        new SpeechSynthesisUtterance(message);

    speech.lang = 'en-IN';
    speech.rate = 0.9;
    speech.pitch = 1;
    speech.volume = 1;

    speech.onstart = () => {
        assistantStatus.textContent =
            'AURA is speaking...';
    };

    speech.onend = () => {
        if (!isListening) {
            assistantStatus.textContent =
                'Ready for your command';
        }
    };

    speech.onerror = (event) => {
        console.error(
            'Speech synthesis error:',
            event.error
        );

        assistantStatus.textContent =
            'Speech could not be played.';
    };

    window.speechSynthesis.speak(speech);
}


/* =========================================
   STOP SPEAKING
========================================= */

function stopSpeech() {
    if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
    }

    lastResponse = 'Speech stopped.';

    assistantResponse.textContent =
        'Speech stopped.';

    assistantStatus.textContent =
        'Ready for your command';
}

function readPage() {
    const mainContent =
        document.querySelector('.assistant-main');

    if (!mainContent) {
        respond('I could not find the page content to read.');

    const pageText = mainContent.innerText
        .replace(/\s+/g, ' ')
        .trim();

    lastResponse = pageText;

    assistantResponse.textContent =
        'Reading the current page aloud.';

    speak(pageText);
}

function readSettings() {
    try {
        return JSON.parse(localStorage.getItem('auraSettings') || '{}');
    } catch {
        return {};
    }
}

function readAccessibility() {
    try {
        return JSON.parse(localStorage.getItem('auraAccessibility') || '{}');
    } catch {
        return {};
    }
}

function isEnabled(value, fallback = true) {
    if (value === undefined || value === null) return fallback;
    return value === true || value === 'true';
}

function voiceEnabled() {
    const settings = readSettings();
    return isEnabled(settings.voiceEnabled, true);
}

function speechEnabled() {
    const settings = readSettings();
    const accessibility = readAccessibility();

    return isEnabled(settings.speechEnabled, true) &&
        isEnabled(accessibility.speechEnabled, true);
}

}

function resetListeningUI() {

function startVoiceAssistant() {
    if (!voiceEnabled()) {
        assistantStatus.textContent = 'Voice commands are disabled in Settings.';
        return;
    }

    if (isListening) {
        recognition?.stop();
        return;
    }

    const SpeechRecognition =
        window.SpeechRecognition || window.webkitSpeechRecognition;

    if (!SpeechRecognition) {
        assistantStatus.textContent =
            'Voice recognition is not supported. Try Chrome on your computer.';
        return;
    }

    recognition = new SpeechRecognition();
    recognition.lang = 'en-IN';
    recognition.interimResults = false;
    recognition.continuous = false;
    recognition.maxAlternatives = 1;

    recognition.onstart = () => {
        isListening = true;
        activateAssistant.textContent = 'ðŸ”´ Listening...';
        assistantStatus.textContent = 'Listening. Speak your command.';
        assistantStatus.classList.add('assistant-listening');
    };

    recognition.onresult = event => {
        const command = event.results?.[0]?.[0]?.transcript?.trim();

        if (!command) {
            assistantStatus.textContent = 'I did not catch that. Please try again.';
            return;
        }

        assistantStatus.textContent = `You said: "${command}"`;
        handleVoiceCommand(command);
    };

    recognition.onerror = event => {
        console.error('Speech recognition error:', event.error);

        assistantStatus.textContent =
            event.error === 'not-allowed' || event.error === 'service-not-allowed'
                ? 'Microphone access was denied. Allow microphone access and try again.'
                : event.error === 'no-speech'
                    ? 'I did not hear anything. Please try again.'
                    : 'I could not hear that clearly. Please try again.';

        resetListeningUI();
    };

    recognition.onend = resetListeningUI;

    try {
        recognition.start();
    } catch (error) {
        console.error('Unable to start recognition:', error);
        resetListeningUI();
        assistantStatus.textContent = 'Unable to start listening. Please try again.';
    }
}

async function askAI(question) {
    assistantStatus.textContent = 'AURA is thinking...';

    try {
        const response = await fetch('http://localhost:5000/api/chat', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ message: question })
        });

        let data;

        try {
            data = await response.json();
        } catch {
            throw new Error(`The backend returned an invalid response (${response.status}).`);
        }

        if (!response.ok) {
            throw new Error(data.message || `AI request failed (${response.status}).`);
        }

        if (typeof data.answer !== 'string' || !data.answer.trim()) {
            throw new Error('AURA did not receive a valid answer.');
        }

        reply(data.answer.trim());
    } catch (error) {
        console.error('AURA AI error:', error);
        reply('I could not connect to my AI service right now. Please check that the backend is running and try again.');
    }
}

function handleVoiceCommand(command) {
    const text = command
        .toLowerCase()
        .replace(/[^\w\s]/g, '')
        .replace(/\s+/g, ' ')
        .trim();

    const matches = (...phrases) => phrases.some(phrase => text.includes(phrase));

    if (/^(hi|hello|hey|hai)\b/.test(text) ||
        /^(good morning|good afternoon|good evening)\b/.test(text)) {
        const hour = new Date().getHours();

        reply(hour < 12
            ? 'Good morning! I am AURA, your voice assistant. How can I help you today?'
            : hour < 17
                ? 'Good afternoon! I am AURA. What can I do for you?'
                : 'Good evening! I am AURA. How can I help?');
        return;
    }

    if (matches('who are you', 'what are you', 'your name', 'introduce yourself')) {
        reply('I am AURA, the Voice Controlled Smart Assistant for Accessibility. I help people navigate this interface using voice commands and spoken feedback.');
        return;
    }

    if (matches('what can you do', 'how can you help', 'your features', 'what are your features', 'what commands', 'help')) {
        reply('I can answer general questions using my AI service, tell you the date and time, read the dashboard aloud, and open Settings or Accessibility. Ask me a question or give me a navigation command.');
        return;
    }

    if (matches('how are you', 'how are things', 'are you okay')) {
        reply('I am running and ready to help. What would you like to do?');
        return;
    }

    if (matches('thank you', 'thanks a lot', 'thanks')) {
        reply('You are welcome! I am happy to help.');
        return;
    }

    if (matches('goodbye', 'bye bye', 'see you later')) {
        reply('Goodbye! You can return whenever you need assistance.');
        return;
    }

    if (matches('what time', 'tell me the time', 'current time', 'time now')) {
        reply(`The current time is ${new Date().toLocaleTimeString([], {
            hour: '2-digit',
            minute: '2-digit'
        })}.`);
        return;
    }

    if (matches('what is the date', 'todays date', 'date today', 'what day is it', 'today date')) {
        reply(`Today is ${new Date().toLocaleDateString('en-IN', {
            weekday: 'long',
            day: 'numeric',
            month: 'long',
            year: 'numeric'
        })}.`);
        return;
    }

    if (matches('what is aura', 'about aura', 'tell me about aura', 'explain aura')) {
        reply('AURA is a Voice Controlled Smart Assistant for Accessibility. It uses browser speech recognition, command matching, AI-generated answers, and text-to-speech to help users interact with a website more easily.');
        return;
    }

    if (matches('open accessibility', 'accessibility page', 'accessibility settings', 'accessibility options', 'open accessibility center', 'accessible features')) {
        reply('Opening the Accessibility Center.');
        setTimeout(goToAccessibility, 700);
        return;
    }

    if (matches('open settings', 'show settings', 'go to settings', 'change settings', 'my preferences')) {
        reply('Opening Settings.');
        setTimeout(goToSettings, 700);
        return;
    }

    if (matches('show activity', 'recent activity', 'my activity', 'activity history')) {
        document.querySelector('#activity').scrollIntoView({ behavior: 'smooth' });
        reply('Here is your recent AURA activity.');
        return;
    }

    if (matches('read this page', 'read the page', 'read dashboard', 'read this dashboard', 'read everything', 'read aloud')) {
        const content = document.querySelector('.dashboard-content');
        if (content) reply(content.innerText);
        else reply('I could not find the dashboard content to read.');
        return;
    }

    if (matches('stop speaking', 'stop talking', 'be quiet', 'silence', 'stop reading')) {
        stopSpeaking();
        assistantStatus.textContent = 'Speech stopped.';
        return;
    }

    if (matches('repeat that', 'say that again', 'repeat yourself')) {
        reply('I am AURA. You can ask general questions or tell me to open Settings or Accessibility.');
        return;
    }

    if (matches('go home', 'open home', 'back to dashboard', 'show dashboard', 'open dashboard')) {
        window.scrollTo({ top: 0, behavior: 'smooth' });
        reply('You are on the AURA dashboard.');
        return;
    }

    askAI(command);
}
